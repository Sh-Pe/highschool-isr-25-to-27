	\subsection{שאלה 1}
	נוכיח ש־$\Delta x = \frac{L \lg}{d}$ בעזרת קירוב לזוויות קטנות. 
	\begin{proof}
		תחת ההנחה ש־$L \gg d$, ניתן להשתמש בטריגו על שני משולשים כדי להבחין בשתי הנוסחאות: 
		\begin{align*}
			(1)&\quad\quad\sin \theta_k = \frac{k \lg}{d} \\
			(2)&\quad\quad\tan \theta_k = \frac{x_k}{L}
		\end{align*}
		באמצעות קירוב זווית קטנה, נסיק ש־$\sin \theta_k \approx \tan \theta_k$. לכן: 
		\[ \frac{k\lg}{d} = \sin \theta_k \approx \tan \theta_k = \frac{x_k}{L} \]
		נתבונן בהפרש $\Delta x_k$ בין $x_k$ ו־$x_{k +1}$: 
		\[ \Delta x_k = x_{k + 1} - x_k = (k + 1)\frac{L\lg}{d} - k \cdot \frac{L\lg}{d} = \frac{L\lg }{d}\cdot\underbrace{((k + 1) - k)}_1 = \frac{L\lg}{d}  \]
		נבחין שבעבור זוויות קטנות, אין תלות ב־$k$. סה''כ $\Delta x = \frac{L \lg }{d}$ כנדרש. 
	\end{proof}

	\subsection{שאלה 1}
	נוכיח ש־$\Delta x = L \lg N^*$
	\begin{proof}
		ידוע מנוסחה (1): 
		\[ \sin a_k = k \lg N^* \]
		ידוע מנוסחה (2): 
		\[ \tan a_k = \frac{x_k}{L} \]
		באמצעות קירוב זווית קטנה, נסיק ש־$\sin a_k \approx \tan a_k$. לכן: 
		\[ k\lg N^* = \sin a_k \approx \tan a_k = \frac{x_k}{L} \]
		דהיינו $\Delta x_k = k L \lg N^*$. 
		נתבונן בהפרש $\Delta x_k$ בין $x_k$ ו־$x_{k +1}$: 
		\[ \Delta x_k = x_{k + 1} - x_k = (k + 1)L\lg N^* - kL\lg N^* = L\lg N^*\cdot\underbrace{((k + 1) - k)}_1 = L\lg N^*  \]
		נבחין שבעבור זוויות קטנות, אין תלות ב־$k$. סה''כ $\Delta x = L \lg N^*$ כנדרש. 
	\end{proof}

	\begin{enumerate}[(A)]
		\item נחשב את מרחק המקסימום מסדר ראשון מהאנך האמצעי. 
		
		\textbf{תשובה: }נמיר יחידות מידה ונגלה ש־$N^* = 10^{6}\frac{1}{\mrm{m}}, \ \lg = 4 \cdot 10^{-7}\mrm{m}, L = 2.5\mrm{m}$. נציב בנוסחה: 
		\[ \Delta x_1 = L \cdot \frac{\lg}{d} = {2.5 \cdot 4 \cdot 10^{-7} \cdot 10^{6}} = \bm{1 \mrm{m}} \]
		
		\item נסמן את מספר קווי המקסימום ב־$N$. ידוע $n_{\max} \le \frac{d}{\lg}$, כלומר: 
		\begin{align*}
			N &= 2n_{\max} + 1 \\
			&= 2 \cdot \left\lfloor \frac{d}{\lg} \right\rfloor + 1 \\
			&= 2 \cdot \left\lfloor \frac{1}{N^*\lg} \right\rfloor + 1 \\
			&= 2 \cdot \left\lfloor \frac{1}{10^{6} \cdot 4 \cdot 10^{-7}} \right\rfloor + 1 = 2 \cdot 2 + 1 = \bm{5}
		\end{align*}
		\item אם אותה המערכת תהיה בתוך שמן בעל מקדם שבירה $1.6$, נבחין שמספר קווי המקסימום יהיה 
		
		\textbf{תשובה. }לעשות
		
		\item התלמיד מנסה לקבל במרחק $2.5\mrm{m}$ תבנית התאבכות דרך סריג עם קרני X (אורך גל $0.1\mrm{nm} \le \lg \le 1\mrm{mm}$), אך אינו מצליח. נסביר את הסיבה. 
		
		\textbf{תשובה: }בהתאם לנוסחה הקודמת, נקבל שכמות קווי המקסימום תהיה:

		כלומר, אותו האור מתחלק על 2001 קווי מקסימום על המסך. זוהי כמות גדולה של קווי מקסימום, מעבר לכושר הראייה של התלמיד, ולכן הוא לא יצליח לראות תבנית התאבכות.

		\item נסביר בעזרת ביטוי מתאים שעבור זוויות קטנות, $X_1 \propto L$. 
		
		\textbf{תשובה. }ע''פ נוסחת יאנג בעבור זוויות קטנות כאשר $\sin x \approx \tan x$, שמביאה לנו את הנוסחה $X_1 = N^*L\lg$, ומפני ש־$N^*$ ו־$\lg$ קבועים, יש יחס ישר בין $X_1$ לבין $L$. 
		\item נסרטט גרף המקשר בין שני הנתונים בטהל: 
		
		\begin{figure}[h]
			\centering
			\includegraphics[width=0.5\linewidth]{images/img1}
			\label{fig:img1}
		\end{figure}
	
		\item ניעזר בגרף כדי למצוא את קבוע הסריג. נבחין שהשיפוע הוא $0.05 = m = N^*\lg $. ידוע $\lg = 400\mrm{nm}$ ולכן קבוע הסריג הוא: 
		\[ \frac{0.05}{400 \cdot 10^{-9}} = 125,000 \csb{\frac{1}{\mrm{m}}} = 125\csb{\frac{1}{\mrm{mm}}} \]
		
		\item בניסוי אחר, הואר הסריג בעל הקבוע $200\csb{\frac{1}{\mrm{mm}}}$ בשני צבעים בו זמנית: סגול באורך $\lg_1 = 400\mrm{nm}$ ואדום באורך $\lg_2 = 600\mrm{nm}$. במסך הוצב במרחק $2.5\mrm{m}$ מהסריג המואר. 
		\begin{itemize}
			\item \textbf{הצבע המופיע בתבנית: }נבחין ש־$X_1 = 30\mrm{cm}$ במקרה זה. 
			
			נסמן ב־$X_1^{(1)}$ את המרחק מנק' המקסימום הראשון בעבור הסגול, וב־$X_1^{(2)}$ בעבור האדום. 
			\begin{gather*}
				X_1^{(1)} = \lg_1N^*L =  400 \cdot 10^{-9} \cdot 200 \cdot 10^{3} \cdot 2.5 = 0.2\mrm{m} \\
				X_1^{(2)} = \lg_2N^*L =  600 \cdot 10^{-9} \cdot 200 \cdot 10^{3} \cdot 2.5 = 0.3\mrm{m}
			\end{gather*}
			קל להבחין מהסרטוט שמתרחש $X_1=0.3\mrm{m}$. לכן $X_1 = X_1^{(1)}$ כלומר האור שאליו שייכת תבנית ההתאבכות הוא האדום. 
			\item \textbf{התבנית עם הצבע השני: }הסרטוט הנ''ל מכיל את שני הצבעים, באותו קנה מידה: 
			
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.5\linewidth]{images/img2}
				\label{fig:img2}
			\end{figure}
			
			\item בניסוי אחר שלושה תלמידים לקחו שלוש לוחיות שונות. א' סריג עקיפה, ב' לוחית עם שני סדקים וג' לוחית עם סדק אחד. 
			
			\begin{itemize}
				\item עבור התלמיד ג' עם הסדק היחיד, נחפש גרף שאורך הפס שלו גדול פי 2 מאורך הפסים שלידו. נבחין שגרפים 1 ו־2 בעלי אורך פס קבוע ולכן נסיק שלתלמיד ג' שייך גרף 1. אנו יודעים שככל שיש יותר חריצים בסריג/תבנית כך התמונה של קווי האור הבודדים תתבההר, ולכן גרף 3 מתאים לתלמיד א' וגרף 2 לתלמיד ב'. 
			\end{itemize}
		\end{itemize}

	\end{enumerate}
	\subsection{שאלה 5}
	שני מקורות נקודתיים $S_1, S_2$ יוצרים במים גלים באורך $\lg$. המרחק בין המקורות הוא $d = 10\mrm{cm}$. תדירות כל אחד מהמקורות היא $10\mrm{Hz}$. 
	\begin{enumerate}[(A)]
		\item נסביר מדוע אורך הגל שיוצר כל מקור הוא $2.5\cm$. 
		
		\textbf{תשובה: }ניתן לספור בין נקודה $S_1$ לנקודה $S_2$ ארבעה מחזורי גל. לכן נחלק את $d$ במספר המופעים ונקבל $\frac{10\cm}{4} = 2.5\cm$. 
		\item נחשב את מהירות ההתפשטות של הגלים במים. 
		
		\textbf{תשובה. }$v = \lg f$, וידוע $f =10\mrm{Hz}, \lg = 2.5\cm = 0.025\mrm{m}$. סה''כ: 
		\[ v = \lg f = 0.025 \cdot 10 = \bm{0.25\csb{\frac{\mrm m}{\mrm s}}} \]
		\item מהם התנאים שבהם $P$ שנמצאת במרחק $L = 50\cm$ מול האנך המרכזי, תהיה נקודת מקסימום? 
		
		\textbf{תשובה: }מכיוון שהיא על האנך האמצעי $n = 0$, בהכרח $\Delta r = \sof{S_1P - S_2P} = 0$. ההתאבכות המרכזית (דרגה $0$) תמיד תהיה בנקודה $P$. לכן, רק נדרוש ש־: 
		\begin{itemize}
			\item שני המקורות יהיו קוהרנטיים
			\item הפרש המופע יהיה זהה (הם יהיו ''מסונכרנים`` ביחס לזמן)
		\end{itemize}
		\item \begin{enumerate}[(1)]
			\item נמצא את המרחק $x_1$ בו תיווצר התאבכות בונה מסדר $1$, בהנחה שהזוויות קטנות. 
			
			\textbf{תשובה: }התאבגות בונה מסדר $n = 1$, לפי קירוב זוויות קטנות, תהיה במרחק: 
			\[ x_1 \approx \frac{1\cdot L\lg}{d} = \frac{1 \cdot 0.025 \cdot 0.5}{0.1} = 0.125\mrm m \]
			
			\item המשרעת בכל מקור בנפרד היא $0.5\cm$. נשרטט גרף המתאר את הפונקציה $P$ כפונקציה של הזמן מ־$t = 0$ עד $t = T$, עם ערכים מספריים על הצירים. 
			
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.5\linewidth]{images/img3}
				\label{fig:img3}
			\end{figure}
			
			חישובים לגרף: נחשב מספר נקודות ע''פ זמן מחזור $T$, ונחזור עליהן ממחזוריות. בנק' המקסימום $y = 1\cm$ וזה קורה בהתאבכות בונה מלאה, לפי התנאי $\Delta r = n \lg$, כלומר ב־$t = \frac{0.1}{4} = 0.0255\mrm s$. ב־$t = 0$ אין גלים ו־$y = 0$, ב־$y = -1\cm$ (התאבכות הורסת מקסימלית) באופן דומה $t = \frac{3}{4} \cdot 0.1 = 0.0755$, וב־$y = 0\cm$ התאבכות הורסת מקסימלית $t = T = 0.1, y = 0\cm$. 
			
			
			
		\end{enumerate}
		\item אם יחליפו את המים שבאמבט במים מלוחים בעלי מקדם שבירה גדול יותר, התדירות לא תשתנה: היא נקבעת אך ורק על פי המקור. בנוסף, אלו גלי מים ולא אור, ולכן אין השפעה על מקדם השבירה. סה''כ אין שינוי במהירות, תדירות או אורך הגל. 
		
	\end{enumerate}
