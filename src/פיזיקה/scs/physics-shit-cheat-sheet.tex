		\section{Mechanics}
		\[ \vec v = \frac{\ds}{\dt} \]
		\[ \vec a = \frac{\dv}{\dt} = \frac{\md^{2}s}{\md^{2}t} \]
        Where $\bar v$ is the average velocity:
		\[ \bar v = \frac{\Dg x}{\Dg t} \]
        Assuming acceleration is constant:
		\begin{align*}
			\vec v(t) &= \vec v_0 + \vec at \\
			\vec x(t) &= \vec x_0 + \vec v_0t + \frac{1}{2}\vec at^2 \\
			\vec x(t) &= \vec x_0 + \frac{\vec v_0 + v}{2}t \\
			\norm{v} &= \sqrt{v_0^2 + 2\norm a\norm {x - x_0}} 
		\end{align*}
		
		\section{Forces}
		\[ \vec F_g = m\vec g \]
		\[ \norm{\vec F_{sp}} = k \, \Dg \ell \]
		\[ \norm{f_s} \le \norm{\mu_s \vec N} \]
		\[ \vec f_k = \mu_k \vec N \]
        \textbf{Newton's laws: }
        \begin{enumerate}
            \item A body remains at rest, or in motion at a constant speed in a straight line, unless it is acted upon by a force. 
            \item \hfil $ \sum \vec F = m \vec a $
            \item If two bodies exert forces $\vec F_1, \vec F_2$ on each other, then $\vec F_1 = -\vec F_2$.
        \end{enumerate}
        
		\section{Energy}
		\[ W = \int \vec F(s) \ds \]
		\begin{gather*}
			(\exists c, \tg\, \forall t \co |F(t)| = c, \, \measuredangle F(t) = \tg) \\
			\implies W = F_x \cdot \Dg x = F\cost \Dg s
		\end{gather*}
		\[ E_k = \frac{1}{2}m\norm{v}^2 \]
		\[ U_g = mgh \]
		\[ U_{sp} = \frac{1}{2}k(\Dg \ell)^2 \]
		\[ {E_k}_1 + {U_g}_1 = {E_k}_2 + {U_g}_2 \]
        \[ \textstyle \forall i, j \co \, \cl{\sum E}_i = \cl{\sum E}_j \]
		\[ W_F = \Dg E = E_{\mathrm{final}} - E_{\mathrm{begining}} \]
        
        \columnbreak
		
		\section{Rotational Movement}
		\[ f := \frac{1}{T} \quad \csb{\mathrm{Hz}} \]
		\[ L = r \psi_{\mathrm{rad}} \]
		\[ \wg := 2\pi \. f = \frac{2\pi}{T} \]
		\[ v \peq \frac{2\pi r}{T} \]
		\[ \bar \wg = \frac{\Dg \theta}{\Dg t} \]
		\[ v = \wg r \]
		\[ \norm{\vec a_R} = \frac{\norm{v}^2}{r} = \wg^2 r \]
		\[ P = 2\pi r \]
		Critical Velocity at max.: 
		\[ N = 0 \iff v = \sqrt{gr} \]
		\[ \vec a_T = -g \sin \ag \]
		\[ \vec a := \vec a_T + \vec a_r \]
		\[ \norm a = \sqrt{a_T^2 + a_R^2} \]
		\[ \tant = \frac{|a_T|}{|a_R|} \]
        
		\section{Gravity}
		\[ \cl{\frac{\bar r_1}{\bar r_2}} ^3= \cl{\frac{T_1}{T_2}}^2 \]
		\[ F_g = G \, \frac{Mm}{r^2} \]
		\[ U_G = - \frac{G Mm}{r} \]
        \[ (W_g)_{A \to B} \peq GMm\cl{\frac{1}{r_A} - \frac{1}{r_B}} \]
        \[ \rho = \frac{m}{v} \]
		In a circular motion: 
        \[ E_k = \frac{GMm}{2r} = - \frac{U_G}{2} \]
        \begin{align*}
            \textstyle \sum E_{\text{mechanic}} \displaystyle &= E_k + U_G \\
            &=  -\frac{GMm}{2r} = -E_k
        \end{align*}
        \[ T \peq \frac{2\pi r}{v} \peq 2\pi\sqrt{\frac{r^3}{GM}} \]
        
        \columnbreak
        
        \textbf{Kepler's laws of planetary motion: }
        \begin{enumerate}
            \item The orbit of a planet is an ellipse
            \item A line segment joining a planet and the Sun sweeps out equal areas during equal intervals of time.
            \item For a given gravitational system: 
            \[ \exists c \, \forall i \co \frac{T_i^2}{r_i^3} = c \peq \frac{4\pi^2}{GM} \]
        \end{enumerate}
		
		\section{Momentum}
		\[ \vec p = m \vec v \quad \csb{N \sec} \sim \csb{\frac{\mathrm{kg}\,m}{\sec}}  \]
		\[ \vec J = \int F \dt \seq \vec F \cdot \Dg \vec t = \sof{\vec F} \sof{\Dg \vec t} \cos\theta \]
		\[ \vec J_{\Sigma F} = \sum_{i = 1}^{n}\vec J_{F_i} = \Dg \vec p \]
		\[ \forall t_1, t_2 \in \R \co \sum_{i = 1}^{n} \vec p_i(t_1) = \sum_{i = 1}^{n}\vec p_i(t_2) \]
		In an inelastic collision: 
		\[ Q = \Dg E_k \]
		In an elastic collision, where $v_i$ before collision and $u_i$ after it: 
		\[ \vec v_1 - \vec v_2 = -(\vec u_1 - \vec u_2) \]
        
        Elastic Collision iff no loss of kinetic energy, Inelastic iff not ecstatic.

		\section{Electricity}
		Coulomb's law: 
		\[ \norm{\vec F_E} = k \frac{\sof{q_1} \sof{q_2}}{r^{2}} \]
		Definition of the electrical field of a given particle with charge $q$: 
		\[ \norm{\vec E} = \frac{\vec F_E}{q_B} = k\frac{q}{r^{2}} \csb{\frac{N}{C}} \]
		Definition of charge density: 
		\[ \sg = \frac{Q}{A} \]
		Given an infinity large plane: 
		\[ E = 2\pi k \sg = \frac{\sg}{2\eg_0} \]
		\begin{align*}
			\Delta W_{A \to B} &= kq_1q_2 \cl{\frac{1}{r_B} - \frac{1}{r_A}} \\
			&= (U_E)_B - (U_E)_A
		\end{align*}
		Hence the potential electric energy: 
		\[ U_E = k \sum_{B} \frac{q_Aq_B}{\dist(A, B)} \ \cl{\text{singular }\frac{kqQ}{r}} \]
		Definition of electric potential in a given point: 
		\[ V_A = \sum_B k\frac{q_B}{\dist(A, B)} \csb{V = \frac{J}{C}} \]
		\[ W_{A \to B} = qV_{AB} = q\underbrace{(V_B - V_A)}_{\text{Voltage}} \]
		\[ E = -\frac{\md V}{\dx} \]
		Potential inside a conductive ball: 
		\[ V = \frac{kQ}{R} \]
		\[ E = -\frac{\dv}{\md r} \]
		
		\section{Capacitors}
		Where $V$ is the voltage between both ends of the capacitor, the capacitance is defined as: 
		\[ C:= \frac{q}{V}\csb{F = \frac{Q}{V} = \frac{J}{V^{2}}} \implies q = CV \]
		\[ V = E \cdot d \]
		\[ U = \frac{1}{2}QV = \frac{1}{2}CV^2 \]
		\[ C = \frac{\eg_0 \eg_r A}{d} \]
		\[ E = \frac{\sg}{\eg_0} = \frac{V}{d} \]
